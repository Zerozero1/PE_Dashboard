SELECT dt_referencia, tipo_anomalia, dimensao_afetada,
       SUM(valor_observado) AS observado,
       SUM(valor_esperado) AS media_esperada,
       ROUND(SUM(valor_observado) / NULLIF(SUM(valor_esperado), 0), 1) AS desvio
FROM prescricao.fato_auditoria_dia
WHERE dt_referencia BETWEEN '2026-09-15' AND '2026-09-21'
  AND tipo_anomalia = 'AN1'
GROUP BY 1, 2, 3 ORDER BY 1;
